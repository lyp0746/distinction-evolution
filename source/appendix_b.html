B.5　可运行代码

import numpy as np
def lap(n, E):
    A = np.zeros((n, n))
    for u, v in E: A[u, v] = A[v, u] = 1
    return np.diag(A.sum(1)) - A
def R(L):
    ev = np.sort(np.linalg.eigvalsh(L))
    return ev[-1] / ev[1]
grid = lambda n: (n*n, [((i*n+j), (i+1)*n+j) for i in range(n-1) for j in range(n)]
                       + [((i*n+j), i*n+j+1) for i in range(n) for j in range(n-1)])
for n in (10, 20, 40, 80):
    N, E = grid(n)
    print(n, round(R(lap(N, E)), 1), round(8/np.pi**2*n*n, 1))
